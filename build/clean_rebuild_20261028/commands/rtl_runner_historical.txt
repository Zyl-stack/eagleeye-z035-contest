from pathlib import Path
import subprocess
import sys

def run(args, cwd=None):
    with log.open('a', encoding='utf-8') as f:
        f.write('\n$ ' + ' '.join(str(x) for x in args) + '\n')
        f.flush()
        p = subprocess.run(args, cwd=cwd, stdout=f, stderr=subprocess.STDOUT)
        f.write('EXIT=' + str(p.returncode) + '\n')
    if p.returncode:
        raise SystemExit(p.returncode)

def linux(command, sub=''):
    run(['wsl', '-d', 'Ubuntu-20.04', '--', 'bash', '-lc', 'cd ' + sim + sub + ' && ' + command])

def compare(name, prefix=None):
    args = [sys.executable, '-B', str(root / 'tools/compare.py'), name]
    if prefix:
        args.extend(['--prefix', prefix])
    run(args)
