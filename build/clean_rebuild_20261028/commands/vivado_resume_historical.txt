set ee_out D:/ee_github_build_20261028
set ee_run_dir [file join $ee_out vivado_work eagleeye_z035.runs impl_1]
if {![file isfile [file join $ee_run_dir z100_top_physopt.dcp]]} {
    error "Post-placement checkpoint missing; refusing to reset or resume"
}
open_project [file join $ee_out vivado_work eagleeye_z035.xpr]
set run [get_runs impl_1]
puts "BEFORE_RESUME STATUS=[get_property STATUS $run] PROGRESS=[get_property PROGRESS $run]"

launch_runs impl_1 -to_step write_bitstream -jobs 4

set status [get_property STATUS $run]

report_timing_summary -report_unconstrained -file [file join $reports timing_summary.rpt]

report_methodology -file [file join $reports methodology.rpt]

write_hw_platform -fixed -include_bit -force -file [file join $ee_out eagleeye_z035.xsa]
puts "EAGLEEYE_BUILD_COMPLETE bit=$bit xsa=[file join $ee_out eagleeye_z035.xsa]"
puts "NOTE: Review timing coverage, CDC and DRC; this is not board validation."
